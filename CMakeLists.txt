cmake_minimum_required(VERSION 3.15)
project(auction_tcp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED)
find_package(PostgreSQL REQUIRED)
add_library(common src/network_utils.cpp)
target_include_directories(common PUBLIC include)
target_compile_options(common PRIVATE -Wall -Wextra -Wpedantic)
add_executable(auction_server src/server.cpp src/persistence.cpp src/persistence_pg.cpp src/logger.cpp)
target_include_directories(auction_server PRIVATE include)
target_link_libraries(auction_server PRIVATE common OpenSSL::Crypto PostgreSQL::PostgreSQL)
add_executable(auction_client tools/client.cpp)
target_link_libraries(auction_client PRIVATE common)
target_include_directories(auction_client PRIVATE include)
foreach(target auction_server auction_client)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endforeach()
